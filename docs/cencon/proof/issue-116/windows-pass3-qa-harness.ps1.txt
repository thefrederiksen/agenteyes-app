<#
  QA harness (issue #116 pass 3, Windows QA). Derived from scripts\windows\tray-quit-repro.ps1, extended:
    -Window tray   : start with --tray (no window)
    -Window open   : start without --tray (main window open) and Quit with the window open
    -Window closed : start without --tray, close the main window to the tray (WM_CLOSE to our own window), then Quit
    -Dismiss N     : open the menu and dismiss it N times before the Quit open
    -DismissVia    : wmclose | uia  (UIA WindowPattern.Close, what the previous QA seat used)
    -WalkItems     : walk the menu's UIA items before each dismiss (the previous QA seat's long sequence)
    -NoQuit        : after the dismissals, do not click Quit - check alive, then quit (counts the dismissals only)
  Every message goes to a window owned by the process this script started. /status must read idle first.
  One RESULT line per run.
#>
param(
    [Parameter(Mandatory)][string]$Exe,
    [ValidateSet('tray','open','closed')][string]$Window = 'tray',
    [int]$Dismiss = 0,
    [ValidateSet('wmclose','uia')][string]$DismissVia = 'wmclose',
    [switch]$WalkItems,
    [switch]$NoQuit,
    [string]$Tag = ''
)
$ErrorActionPreference = 'Stop'
Add-Type -AssemblyName UIAutomationClient, UIAutomationTypes
if (-not ('QaTray' -as [type])) {
Add-Type -TypeDefinition @'
using System;
using System.Collections.Generic;
using System.Runtime.InteropServices;
using System.Text;
public static class QaTray
{
    public delegate bool EnumProc(IntPtr h, IntPtr l);
    [DllImport("user32.dll")] public static extern bool EnumWindows(EnumProc f, IntPtr l);
    [DllImport("user32.dll")] public static extern IntPtr FindWindowEx(IntPtr parent, IntPtr after, string cls, string title);
    [DllImport("user32.dll")] public static extern uint GetWindowThreadProcessId(IntPtr h, out uint pid);
    [DllImport("user32.dll", CharSet = CharSet.Unicode)] public static extern int GetClassName(IntPtr h, StringBuilder s, int n);
    [DllImport("user32.dll")] public static extern bool IsWindowVisible(IntPtr h);
    [DllImport("user32.dll")] public static extern bool PostMessage(IntPtr h, uint m, IntPtr w, IntPtr l);
    [DllImport("user32.dll")] public static extern IntPtr GetForegroundWindow();
    [StructLayout(LayoutKind.Sequential)] public struct POINT { public int X, Y; }
    [DllImport("user32.dll")] public static extern bool ScreenToClient(IntPtr h, ref POINT p);
    public static string ClassOf(IntPtr h) { var sb = new StringBuilder(256); GetClassName(h, sb, 256); return sb.ToString(); }
    public static List<IntPtr> WindowsOf(uint pid)
    {
        var list = new List<IntPtr>();
        EnumWindows((h, l) => { uint p; GetWindowThreadProcessId(h, out p); if (p == pid) list.Add(h); return true; }, IntPtr.Zero);
        IntPtr after = IntPtr.Zero;
        while ((after = FindWindowEx(new IntPtr(-3), after, null, null)) != IntPtr.Zero)
        { uint p; GetWindowThreadProcessId(after, out p); if (p == pid) list.Add(after); }
        return list;
    }
}
'@
}
$WM_TRAYMOUSE = 0x800; $WM_RBUTTONUP = 0x205; $WM_CLOSE = 0x10
$WM_MOUSEMOVE = 0x200; $WM_LBUTTONDOWN = 0x201; $WM_LBUTTONUP = 0x202
$PipeName = "agenteyes-app-$($env:USERNAME.ToLowerInvariant())"
function Stamp([string]$m) { Write-Output ("[{0:HH:mm:ss.fff}] {1}" -f (Get-Date), $m) }
function PipePresent { @([System.IO.Directory]::GetFiles('\\.\pipe\') | Where-Object { $_ -like "*\$PipeName" }).Count -gt 0 }
function Lparam([int]$x, [int]$y) { [IntPtr](($y -shl 16) -bor ($x -band 0xFFFF)) }
function Aid([IntPtr]$h) { try { [System.Windows.Automation.AutomationElement]::FromHandle($h).Current.AutomationId } catch { $null } }
function Get-MessageWindow([uint32]$procId) {
    foreach ($h in [QaTray]::WindowsOf($procId)) { if ([QaTray]::ClassOf($h) -like '*AvaloniaMessageWindow*') { return $h } }
    return [IntPtr]::Zero
}
function Get-Popups([uint32]$procId) {
    $found = @()
    foreach ($h in [QaTray]::WindowsOf($procId)) {
        if (-not [QaTray]::IsWindowVisible($h)) { continue }
        $a = Aid $h
        if ($a -like 'AvaloniaTrayPopupRoot*') { $found += $h }
    }
    return , $found
}
function Get-MainWindow([uint32]$procId) {
    foreach ($h in [QaTray]::WindowsOf($procId)) {
        if (-not [QaTray]::IsWindowVisible($h)) { continue }
        $a = Aid $h
        if ($null -ne $a -and $a -notlike 'AvaloniaTrayPopupRoot*') {
            $name = try { [System.Windows.Automation.AutomationElement]::FromHandle($h).Current.Name } catch { '' }
            if ($name -like '*AgentEyes*') { return $h }
        }
    }
    return [IntPtr]::Zero
}
function Open-Menu([uint32]$procId, [IntPtr]$msgWnd) {
    [void][QaTray]::PostMessage($msgWnd, $WM_TRAYMOUSE, [IntPtr]1, [IntPtr]$WM_RBUTTONUP)
    for ($i = 0; $i -lt 300; $i++) {
        Start-Sleep -Milliseconds 100
        $p = Get-Popups $procId
        if ($p.Count -ge 1) {
            Start-Sleep -Milliseconds 150
            $fg = [QaTray]::GetForegroundWindow()
            $script:fgNotes += $(if ($fg -eq $p[0]) { 'FG' } else { 'notFG' })
            return $p[0]
        }
    }
    throw "the tray menu popup did not appear within 30 s"
}
function Get-Items([IntPtr]$popup) {
    $el = [System.Windows.Automation.AutomationElement]::FromHandle($popup)
    $cond = New-Object System.Windows.Automation.PropertyCondition([System.Windows.Automation.AutomationElement]::ControlTypeProperty, [System.Windows.Automation.ControlType]::MenuItem)
    for ($i = 0; $i -lt 100; $i++) {
        $items = @($el.FindAll([System.Windows.Automation.TreeScope]::Descendants, $cond))
        if ($items.Count -ge 4) { return , $items }
        Start-Sleep -Milliseconds 100
    }
    throw "menu items did not reach the UIA tree within 10 s"
}
function Click-Quit([IntPtr]$popup) {
    $quit = $null
    foreach ($item in (Get-Items $popup)) { if ($item.Current.Name -like 'Quit AgentEyes*') { $quit = $item } }
    if ($null -eq $quit) { throw "no 'Quit AgentEyes' item" }
    $r = $quit.Current.BoundingRectangle
    $pt = New-Object QaTray+POINT
    $pt.X = [int]($r.X + $r.Width / 2); $pt.Y = [int]($r.Y + $r.Height / 2)
    [void][QaTray]::ScreenToClient($popup, [ref]$pt)
    $lp = Lparam $pt.X $pt.Y
    [void][QaTray]::PostMessage($popup, $WM_MOUSEMOVE, [IntPtr]0, $lp); Start-Sleep -Milliseconds 80
    [void][QaTray]::PostMessage($popup, $WM_LBUTTONDOWN, [IntPtr]1, $lp); Start-Sleep -Milliseconds 40
    [void][QaTray]::PostMessage($popup, $WM_LBUTTONUP, [IntPtr]0, $lp)
}
function Dismiss-Menu([uint32]$procId, [IntPtr]$popup) {
    if ($WalkItems) { $names = (Get-Items $popup | ForEach-Object { $_.Current.Name }) -join ' | '; $script:walked = $names }
    if ($DismissVia -eq 'uia') {
        $wp = [System.Windows.Automation.AutomationElement]::FromHandle($popup).GetCurrentPattern([System.Windows.Automation.WindowPattern]::Pattern)
        $wp.Close()
    } else { [void][QaTray]::PostMessage($popup, $WM_CLOSE, [IntPtr]0, [IntPtr]0) }
    for ($i = 0; $i -lt 30; $i++) { Start-Sleep -Milliseconds 100; if ((Get-Popups $procId).Count -eq 0) { return } }
    throw "popup still visible 3 s after dismiss"
}

# --- preconditions: the owner's recorder must be idle; no other desktop-app instance; no pipe
$st = (Invoke-RestMethod -Uri 'http://127.0.0.1:7882/status' -TimeoutSec 5).State
if ($st -ne 'idle') { Write-Output "RESULT: ABORT recorder state=$st (not idle) - not launching"; exit 2 }
for ($w = 0; $w -lt 40 -and (Get-Process AgentEyesDesktop -ErrorAction SilentlyContinue); $w++) { Start-Sleep -Milliseconds 500 }
if (Get-Process AgentEyesDesktop -ErrorAction SilentlyContinue) { Write-Output "RESULT: ABORT an AgentEyesDesktop is already running"; exit 2 }
if (PipePresent) { Write-Output "RESULT: ABORT pipe $PipeName exists before start"; exit 2 }

$script:fgNotes = @(); $script:walked = ''
$logFile = Join-Path $env:LOCALAPPDATA ("AgentEyes\logs\AgentEyes-{0:yyyyMMdd}.log" -f (Get-Date))
$argList = if ($Window -eq 'tray') { @('--tray') } else { @() }
$proc = if ($argList.Count) { Start-Process -FilePath $Exe -ArgumentList $argList -PassThru } else { Start-Process -FilePath $Exe -PassThru }
Stamp "status=$st started pid=$($proc.Id) window=$Window dismiss=$Dismiss via=$DismissVia walk=$WalkItems noQuit=$NoQuit tag=$Tag"
function Crashed { @(Get-Content $logFile | Where-Object { $_ -match "\[pid $($proc.Id)\].*unhandled exception" }).Count -gt 0 }
$stage = 'start'; $crashStage = ''
try {
    $msgWnd = [IntPtr]::Zero
    for ($i = 0; $i -lt 600 -and $msgWnd -eq [IntPtr]::Zero; $i++) {
        Start-Sleep -Milliseconds 100
        if ($proc.HasExited) { throw "exited during start-up code $($proc.ExitCode)" }
        $msgWnd = Get-MessageWindow ([uint32]$proc.Id)
    }
    if ($msgWnd -eq [IntPtr]::Zero) { throw "no AvaloniaMessageWindow after 60 s" }
    Start-Sleep -Milliseconds 1500
    if ($Window -ne 'tray') {
        $mw = [IntPtr]::Zero
        for ($i = 0; $i -lt 300 -and $mw -eq [IntPtr]::Zero; $i++) { Start-Sleep -Milliseconds 100; $mw = Get-MainWindow ([uint32]$proc.Id) }
        if ($mw -eq [IntPtr]::Zero) { throw "main window did not appear" }
        Stamp "main window=0x$($mw.ToString('X')) visible"
        if ($Window -eq 'closed') {
            $stage = 'close-window'
            [void][QaTray]::PostMessage($mw, $WM_CLOSE, [IntPtr]0, [IntPtr]0)
            for ($i = 0; $i -lt 50 -and [QaTray]::IsWindowVisible($mw); $i++) { Start-Sleep -Milliseconds 100 }
            Start-Sleep -Milliseconds 500
            if ($proc.HasExited -or (Crashed)) { throw "went down after closing the window" }
            Stamp "main window closed to tray: visible=$([QaTray]::IsWindowVisible($mw)) alive=$(-not $proc.HasExited)"
        }
    }
    for ($n = 1; $n -le $Dismiss; $n++) {
        $stage = "dismiss$n"
        $popup = Open-Menu ([uint32]$proc.Id) $msgWnd
        Dismiss-Menu ([uint32]$proc.Id) $popup
        Start-Sleep -Milliseconds 1000
        if ($proc.HasExited -or (Crashed)) { $crashStage = $stage; throw "went down after dismiss $n" }
    }
    if ($Dismiss -gt 0) { Stamp "after $Dismiss dismiss(es): alive, no unhandled exception" }
    if ($NoQuit) { Start-Sleep -Seconds 2; if ($proc.HasExited -or (Crashed)) { $crashStage = 'post-dismiss'; throw "went down after dismissals" } }
    $stage = 'quit'
    $popup = Open-Menu ([uint32]$proc.Id) $msgWnd
    Click-Quit $popup
    Stamp "clicked Quit"
    if (-not $proc.WaitForExit(15000)) { Stamp "FAIL: no exit within 15 s" }
}
catch { Stamp "FAIL at stage=${stage}: $($_.Exception.Message)"; if (-not $crashStage) { $crashStage = $stage } }
Start-Sleep -Milliseconds 300
$exitCode = if ($proc.HasExited) { $proc.ExitCode } else { 'running' }
$pipeAfter = PipePresent
$lines = @(Get-Content $logFile | Where-Object { $_ -match "\[pid $($proc.Id)\]" -and $_ -notmatch '\]    at ' })
$need = '[App] Quit (', '[SingleInstance] Dispose', '[Program] Main: exited'
$missing = @($need | Where-Object { $n = $_; -not ($lines | Where-Object { $_.Contains($n) }) })
$crashed = @($lines | Where-Object { $_ -match 'unhandled exception' }).Count -gt 0
$cancelled = @($lines | Where-Object { $_ -match 'cancelled a re-entrant close' }).Count
$observed = @($lines | Where-Object { $_ -match 'QA-CONTROL OBSERVED' }).Count
$selfClosed = (-not ($lines | Where-Object { $_ -match 'DeferredQuit\] Request|\[App\] Quit \(' })) -and (($lines | Where-Object { $_ -match 'closing \(reason=WindowClosing, active=False\)' }).Count -gt 0)
$pass = ($proc.HasExited -and $exitCode -eq 0 -and $missing.Count -eq 0 -and -not $crashed -and -not $pipeAfter)
if (-not $proc.HasExited) { Stop-Process -Id $proc.Id -Force; Stamp "killed own leftover test process $($proc.Id)" }
Write-Output "--- log lines for pid $($proc.Id):"
$lines | Where-Object { $_ -notmatch 'PollAsync|Render:|AppSettings|StartPolling' } | ForEach-Object { Write-Output $_ }
Write-Output ("RESULT: {0} pid={1} tag={2} window={3} dismiss={4} via={5} walk={6} noQuit={7} exit={8} missing=[{9}] unhandled={10} crashStage={11} pipeAfter={12} guardCancelled={13} controlObserved={14} fg=[{15}]" -f `
    $(if ($pass) { 'PASS' } elseif ($crashed) { 'CRASH' } elseif ($selfClosed) { 'INVALID-popup-self-closed' } else { 'FAIL' }), $proc.Id, $Tag, $Window, $Dismiss, $DismissVia, [bool]$WalkItems, [bool]$NoQuit, $exitCode, ($missing -join '; '), $crashed, $(if ($crashed) { $crashStage } else { '-' }), $pipeAfter, $cancelled, $observed, ($script:fgNotes -join ','))
