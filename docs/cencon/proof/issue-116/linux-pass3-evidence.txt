## 1. Tray-quit helper (com.canonical.dbusmenu Event 'clicked' on the Quit entry)
#!/bin/bash
# Click the tray menu's "Quit" entry of the AgentEyes app with the given pid, over com.canonical.dbusmenu.
pid=$1
N=$(gdbus call --session -d org.freedesktop.DBus -o /org/freedesktop/DBus -m org.freedesktop.DBus.ListNames | tr ',' '\n' | grep -o "org.kde.StatusNotifierItem-$pid-[0-9]*")
[ -n "$N" ] || { echo "no SNI for pid $pid"; exit 1; }
M=$(gdbus call --session -d $N -o /StatusNotifierItem -m org.freedesktop.DBus.Properties.Get org.kde.StatusNotifierItem Menu | grep -o "/net[^']*")
L=$(gdbus call --session -d $N -o $M -m com.canonical.dbusmenu.GetLayout -- 0 -1 "['label']")
id=$(echo "$L" | grep -o "([0-9]*, {'label': <'Quit[^']*" | grep -o "^([0-9]*" | tr -d '(')
[ -n "$id" ] || { echo "no Quit item in layout: $L"; exit 1; }
echo "SNI=$N menu=$M quit-item-id=$id"
echo "quit label: $(echo "$L" | grep -o "'Quit[^']*'")"
echo "$(date +%T.%N | cut -c1-12) sending dbusmenu Event clicked to item $id"
gdbus call --session -d $N -o $M -m com.canonical.dbusmenu.Event -- $id clicked "<int32 0>" 0

## 2. Cycle script (criterion 11)
#!/bin/bash
# One criterion-11 cycle: start the app (window opens), close to tray, 'show' handoff, 4 --view handoffs,
# close to tray, click tray Quit over dbusmenu. No kill, no signal.
S=$(dirname "$(dirname "$(readlink -f "$0")")")
APP=${APP:-$S/app/AgentEyesDesktop}
SOCK=/run/user/1000/agenteyes-app.sock
n=$1
ts() { date +%T.%N | cut -c1-12; }
win() { wmctrl -lp | awk -v p=$1 '$3==p {print $1}'; }
mapped() { wmctrl -lp | awk -v p=$1 '$3==p' | wc -l; }
sock() { ls -la --time-style=full-iso $SOCK 2>&1 | tr -s ' '; echo -n " | "; ss -xlp 2>/dev/null | grep "$SOCK" | grep -o 'users:.*' ; }
echo "=== cycle $n start $(ts) load: $(cut -d' ' -f1-3 /proc/loadavg) status=$(curl -s -m3 127.0.0.1:7882/status | grep -o '"State": *"[a-z]*"')"
echo "socket before: $(sock)"
setsid $APP > $S/run$n.stdout 2>&1 < /dev/null &
pid=$!
for i in $(seq 1 200); do w=$(win $pid); [ -n "$w" ] && break; sleep 0.1; done
echo "first instance pid=$pid window=$w at $(ts)"
sleep 1.5
echo "socket while running: $(sock)"
echo "maps: Avalonia.Win32.dll=$(grep -c 'Avalonia.Win32.dll' /proc/$pid/maps) Avalonia.X11.dll=$(grep -c 'Avalonia.X11.dll' /proc/$pid/maps) Avalonia.FreeDesktop.dll=$(grep -c 'Avalonia.FreeDesktop.dll' /proc/$pid/maps)"
[ "$n" = 1 ] && { sleep 3; xwd -silent -id $w -out $S/run1-window.xwd && echo "xwd captured $w"; }
wmctrl -ic $w; sleep 1
echo "after close-to-tray: mapped=$(mapped $pid)"
echo "second launch (show) $(ts)"; $APP; echo " exit=$?"; sleep 1
echo "after second launch: mapped=$(mapped $pid)"
for v in library record library record; do $APP --view $v; echo " handoff '--view $v' exit=$?"; sleep 0.7; done
wmctrl -ic $(win $pid); sleep 1
echo "before quit: mapped=$(mapped $pid) maps: Avalonia.Win32.dll=$(grep -c Avalonia.Win32.dll /proc/$pid/maps) Avalonia.X11.dll=$(grep -c Avalonia.X11.dll /proc/$pid/maps)"
( m=0; c=0; for i in $(seq 1 60); do k=$(mapped $pid); c=$((c+1)); [ $k -gt $m ] && m=$k; sleep 0.05; done; echo "window watcher during quit: $c samples, max mapped=$m" ) &
wp=$!
$S/bin/tray-quit.sh $pid
wait $pid 2>/dev/null; rc=$?
echo "exit=$rc at $(ts)"
sleep 0.2; kill -0 $pid 2>/dev/null && echo "pid $pid alive after quit: YES" || echo "pid $pid alive after quit: NO"
wait $wp
echo "socket after: $(sock) | ss listeners: $(ss -xlp 2>/dev/null | grep -c "$SOCK")"
echo "first instance pid for log: $pid"
