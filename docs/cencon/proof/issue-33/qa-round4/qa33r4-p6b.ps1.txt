function HudHandle() {
    Add-Type -AssemblyName UIAutomationClient, UIAutomationTypes
    $A = [System.Windows.Automation.AutomationElement]
    $c1 = New-Object System.Windows.Automation.PropertyCondition($A::NameProperty, 'Recording HUD')
    $c2 = New-Object System.Windows.Automation.PropertyCondition($A::ControlTypeProperty, [System.Windows.Automation.ControlType]::Window)
    $cond = New-Object System.Windows.Automation.AndCondition($c1, $c2)
    for ($i = 0; $i -lt 30; $i++) {
        $w = $A::RootElement.FindFirst('Children', $cond)
        if ($w) { return [IntPtr]$w.Current.NativeWindowHandle }
        Start-Sleep -Milliseconds 400
    }
    throw 'HUD not found'
}

function Frames($log) {
    if (-not (Test-Path $log)) { return -1 }
    $n = -1
    foreach ($line in Get-Content $log) { $m = [regex]::Match($line, 'frame=\s*(\d+)'); if ($m.Success) { $n = [Math]::Max($n, [int]$m.Groups[1].Value) } }
    return $n
}
