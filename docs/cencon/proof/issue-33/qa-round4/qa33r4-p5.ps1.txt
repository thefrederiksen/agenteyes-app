function Frames($log) {
    if (-not (Test-Path $log)) { return -1 }
    $n = -1
    foreach ($line in Get-Content $log) {
        $m = [regex]::Match($line, 'frame=\s*(\d+)')
        if ($m.Success) { $n = [Math]::Max($n, [int]$m.Groups[1].Value) }
    }
    return $n
}

function Run60($armed, $label) {
    StartApp $(if ($armed) { $ARMED } else { $UNARMED }) $false
    StartChurn 1 110
    Rec
    while ((Status).ElapsedSeconds -lt 62) { Start-Sleep -Seconds 2 }
    StopRec
    Start-Sleep -Seconds 30
    $dir = NewestDir
    $rawlog = Join-Path $dir 'raw.mp4.ffmpeg.log'
    $camlog = Join-Path $dir 'camera.mp4.ffmpeg.log'
    $rec = Join-Path $dir 'recording.mp4'; $cam = Join-Path $dir 'camera.mp4'
    $ds = Dur $rec; $dc = Dur $cam
    Write-Output ("  {0,-14} screenDrops={1,-4} cameraDrops={2,-4} deliveredFrames={3,-6} screenDur={4,-8} cameraDur={5,-8} delta={6:N3}  {7}" -f `
        $label, (Drops $rawlog), (Drops $camlog), (Frames $rawlog), [Math]::Round($ds,3), [Math]::Round($dc,3), [Math]::Abs($ds - $dc), (Split-Path $dir -Leaf))
    Get-Process AgentEyesApp -ErrorAction SilentlyContinue | Stop-Process -Force
    Start-Sleep -Seconds 2
}
