function Hud($a, $b) { if ($b) { & powershell -NoProfile -File "$SP\hud33.ps1" -Action $a -Arg $b } else { & powershell -NoProfile -File "$SP\hud33.ps1" -Action $a } }
function Cfg($what) { Write-Output ("  config {0}: {1}" -f $what, (((Get-Content $cfg | Select-String 'HudWidth|HudHeight|HudLeft|HudTop|HudPreviewVisible') -join ' ') -replace '\s+', ' ')) }
function Post($p, $b) { Invoke-RestMethod -Uri "http://127.0.0.1:7882$p" -Method Post -Body ($b | ConvertTo-Json) -ContentType 'application/json' -TimeoutSec 30 }

function Fresh($json) {
    Get-Process AgentEyesApp -ErrorAction SilentlyContinue | Stop-Process -Force
    Start-Sleep -Seconds 3
    $json | Out-File -FilePath $cfg -Encoding utf8

    for ($i = 0; $i -lt 60; $i++) {

    }
}
