function Hud($a, $b, $o) { if ($o) { & powershell -NoProfile -File "$SP\qa-hud-uia-r3.ps1" -Action $a -Arg $b -Out $o } elseif ($b) { & powershell -NoProfile -File "$SP\qa-hud-uia-r3.ps1" -Action $a -Arg $b } else { & powershell -NoProfile -File "$SP\qa-hud-uia-r3.ps1" -Action $a } }

function Ffmpegs()  {
    $n = 0; $cam = 0
    Get-CimInstance Win32_Process -Filter "Name='ffmpeg.exe'" | ForEach-Object {
        $n++
        if ($_.CommandLine -match '-f dshow' -and $_.CommandLine -match '-i video=') { $cam++ }
    }
    Write-Output ("  ffmpeg processes={0}  holding the CAMERA device={1}" -f $n, $cam)
}
