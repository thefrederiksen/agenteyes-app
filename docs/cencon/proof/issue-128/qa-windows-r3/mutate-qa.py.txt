def run(cmd, env=None, timeout=1800):
    e = dict(os.environ)
    if env: e.update(env)
    return subprocess.run(cmd, cwd=REPO, env=e, capture_output=True, text=True, timeout=timeout, shell=isinstance(cmd, str))


def ffmpeg_count():
    r = run(["tasklist", "/FI", "IMAGENAME eq ffmpeg.exe"])
    return r.stdout.lower().count("ffmpeg.exe")


def diff_empty():
    return run(["git", "diff", "--quiet", "--", "src", "tests"]).returncode == 0


def apply(edits):
    for path, old, new in edits:
        full = os.path.join(REPO, path)
        with open(full, encoding="utf-8", newline="") as f:
            s = f.read()
        if "\r\n" in s:
            old_, new_ = old.replace("\n", "\r\n"), new.replace("\n", "\r\n")
        else:
            old_, new_ = old, new
        n = s.count(old_)
        if n != 1:
            raise RuntimeError(f"{path}: anchor found {n} times: {old[:80]!r}")
        with open(full, "w", encoding="utf-8", newline="") as f:
            f.write(s.replace(old_, new_))


def restore(edits):
    files = sorted({p for p, _, _ in edits})
    run(["git", "checkout", "--"] + files)


out = open(OUT, "a", encoding="ascii", errors="replace")
def w(line):
    print(line, flush=True); out.write(line + "\n"); out.flush()

w(f"# mutation run at {head}, {time.strftime('%Y-%m-%d %H:%M:%S')}, ffmpeg dir {FF}")
w(f"restore baseline: diff empty = {diff_empty()}")
for mid, desc, edits in MUTATIONS:
    if ONLY and mid not in ONLY: continue
    t0 = time.time()
    try:
        apply(edits)
        b = run(["dotnet", "build", "AgentEyes.sln", "-c", "Release"])
        if "0 Error(s)" not in b.stdout:
            errs = [l.strip() for l in b.stdout.splitlines() if " error " in l][:3]
            w(f"{mid} | {desc} | BUILD FAILED | {errs}")
            continue
        t = run(["dotnet", "test", "AgentEyes.sln", "-c", "Release", "--no-build"], env={"QA_RECORD_FFMPEG_DIR": FF})
        summary = [l.strip() for l in t.stdout.splitlines() if l.strip().startswith(("Passed!", "Failed!"))]
        failed = sorted({re.sub(r"\s*\[.*$", "", l.strip()[len("Failed "):]) for l in t.stdout.splitlines()
                         if l.strip().startswith("Failed AgentEyes")})
        w(f"{mid} | {desc} | {'; '.join(s.split(' - ')[0] for s in summary)} | ffmpeg left {ffmpeg_count()} | {int(time.time()-t0)}s")
        for f in failed: w(f"    {f}")
        if not failed: w("    (no test failed - SURVIVES)")
    finally:
        restore(edits)
        w(f"    restored, diff empty = {diff_empty()}")
