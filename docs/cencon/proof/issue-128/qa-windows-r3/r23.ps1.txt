"=== $tag app=$app"
function Show($d) {
  $m = Get-Content "$d\manifest.json" -Raw | ConvertFrom-Json
  "  [$(Split-Path $d -Leaf)]"
  "    PostProcessing: $($m.PostProcessing | ConvertTo-Json -Compress)"
  "    PendingMux set: $($null -ne $m.PendingMux); OriginalFiles: $($m.OriginalFiles -join ',')"
  "    files: $((Get-ChildItem $d | ForEach-Object { $_.Name + '(' + $_.Length + ')' }) -join ', ')"
  if (Test-Path "$d\recording.mp4") { "    recording.mp4 md5: $((Get-FileHash "$d\recording.mp4" -Algorithm MD5).Hash)" } else { "    recording.mp4: ABSENT" }
}

  Start-Sleep 60
  "--- after launch $i"; Show $pub; Show $gone

  "--- log lines from pid $($a.Id) naming either recording, plus FindUnfinished summaries:"
  Get-ChildItem "$env:LOCALAPPDATA\AgentEyes\logs\*.log" | Where-Object { $_.LastWriteTime -gt $t } | ForEach-Object { Get-Content $_.FullName } |

  Stop-Process -Id $a.Id -Force; Start-Sleep 2
}
