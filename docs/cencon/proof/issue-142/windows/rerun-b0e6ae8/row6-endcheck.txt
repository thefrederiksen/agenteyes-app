## The check script (endcheck.ps1)
param(
  [Parameter(Mandatory)] [string]$File,
  [Parameter(Mandatory)] [double]$Expected,   # expected take length in seconds, from the driver's own clock
  [double]$Tolerance = 2.0,
  [double]$Window = 2.0,
  [double]$MinFps = 20.0                      # frames required per second in the end window
)
# Row 6 check, sharpened: duration matches the EXPECTED length AND frames decode AT the expected end.
$ffp = "$env:LOCALAPPDATA\AgentEyes\app\ffprobe.exe"; $ff = "$env:LOCALAPPDATA\AgentEyes\app\ffmpeg.exe"
"file=$(Split-Path $File -Leaf) size=$((Get-Item $File).Length) expected=$Expected s tolerance=+-$Tolerance s window=last $Window s of the expected length"
$dur = & $ffp -v error -select_streams v:0 -show_entries 'format=duration' -of csv=p=0 $File 2>&1 | Select-Object -First 1
$durOk = $false; $d = 0.0
if ([double]::TryParse(("$dur").Trim(), [ref]$d)) { $durOk = [math]::Abs($d - $Expected) -le $Tolerance }
"  duration=$dur  |duration-expected|=$(if ($d) { [math]::Round([math]::Abs($d-$Expected),3) } else { 'n/a' })  -> $(if ($durOk) { 'OK' } else { 'FAIL' })"
& $ff -hide_banner -v error -i $File -f null - 2>$null; "  full decode exit=$LASTEXITCODE (reported, not judged: a truncated file can decode to 0)"
$lo = $Expected - $Window; $hi = $Expected + 0.05
$start = [math]::Max(0, $lo - 3)
$pts = & $ffp -v error -select_streams v:0 -read_intervals "$start%" -show_entries 'frame=pts_time' -of csv=p=0 $File 2>$null |
  ForEach-Object { $x = 0.0; if ([double]::TryParse(("$_").Trim().TrimEnd(','), [ref]$x)) { $x } }
$inWin = @($pts | Where-Object { $_ -ge $lo -and $_ -le $hi })
$need = [math]::Floor($MinFps * $Window)
$endOk = $inWin.Count -ge $need
"  decoded frames with pts in [$([math]::Round($lo,3)), $([math]::Round($hi,3))]: $($inWin.Count) (need >= $need)  last decoded pts=$(if ($pts) { ($pts | Measure-Object -Maximum).Maximum } else { 'none' })  -> $(if ($endOk) { 'OK' } else { 'FAIL' })"
"  VERDICT: $(if ($durOk -and $endOk) { 'PASS' } else { 'FAIL' })"
