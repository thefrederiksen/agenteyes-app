## The measuring script (emptywindow.ps1)
param([Parameter(Mandatory)] [string]$Out, [int]$Seconds = 20)
# Measures when the first byte reaches the capture file, at default settings (30 fps), read-only polling.
$api='http://127.0.0.1:7882'
function W($s) { $l = "{0} {1}" -f (Get-Date -Format 'HH:mm:ss.fff'), $s; $l | Out-File -Append -Encoding ascii $Out; $l }
"" | Out-File -Encoding ascii $Out
W "ffmpeg before: $(@(Get-CimInstance Win32_Process -Filter "Name='ffmpeg.exe'").Count) process(es)"
$tReq = Get-Date
$r = Invoke-RestMethod -Method Post -Uri "$api/record/start" -Body '{"mode":"video","screen":1,"source":"none"}' -ContentType 'application/json'
$tResp = Get-Date
$f = Join-Path $r.Dir $r.CaptureFile
W ("start request {0}, response after {1:N3}s, capture file {2}" -f $tReq.ToString('HH:mm:ss.fff'), ($tResp-$tReq).TotalSeconds, (Split-Path $f -Leaf))
$ff = Get-CimInstance Win32_Process -Filter "Name='ffmpeg.exe'" | Select-Object -First 1
$ffStart = $ff.CreationDate
W ("capture ffmpeg pid={0} ppid={1} created {2}" -f $ff.ProcessId, $ff.ParentProcessId, $ffStart.ToString('HH:mm:ss.fff'))
$first = $null; $prev = -1; $changes = @()
$end = $tReq.AddSeconds($Seconds)
while ((Get-Date) -lt $end) {
  $len = if (Test-Path $f) { (Get-Item $f).Length } else { -1 }
  $now = Get-Date
  if ($len -ne $prev) { $changes += ("  +{0:N3}s from ffmpeg start: {1} bytes" -f ($now-$ffStart).TotalSeconds, $len); $prev = $len }
  if (-not $first -and $len -gt 0) { $first = $now }
  Start-Sleep -Milliseconds 50
}
Invoke-RestMethod -Method Post -Uri "$api/record/stop" | Out-Null
W "stopped"
$changes | Select-Object -First 8 | ForEach-Object { W $_ }
if ($first) {
  W ("FIRST BYTE: {0:N3}s after ffmpeg process creation, {1:N3}s after the start request" -f ($first-$ffStart).TotalSeconds, ($first-$tReq).TotalSeconds)
} else { W "FIRST BYTE: none within $Seconds s" }
W "DIR=$($r.Dir)"
