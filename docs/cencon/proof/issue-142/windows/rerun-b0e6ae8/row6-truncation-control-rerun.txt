## The driver (control.ps1)
$S = Split-Path -Parent $MyInvocation.MyCommand.Path
$src = "$env:USERPROFILE\Videos\AgentEyes\2026-11-16_161739_video"
$ffp = "$env:LOCALAPPDATA\AgentEyes\app\ffprobe.exe"; $ff = "$env:LOCALAPPDATA\AgentEyes\app\ffmpeg.exe"
function Atoms($f) {
  $fs = [IO.File]::OpenRead($f); $br = New-Object IO.BinaryReader($fs); $out = @()
  try { while ($fs.Position + 8 -le $fs.Length) {
      $b = $br.ReadBytes(4); [Array]::Reverse($b); $sz = [BitConverter]::ToUInt32($b,0)
      $t = [Text.Encoding]::ASCII.GetString($br.ReadBytes(4)); $hdr = 8
      if ($sz -eq 1) { $b = $br.ReadBytes(8); [Array]::Reverse($b); $sz = [BitConverter]::ToUInt64($b,0); $hdr = 16 }
      if ($sz -eq 0) { $sz = $fs.Length - $fs.Position + $hdr }
      $start = $fs.Position - $hdr; $out += [pscustomobject]@{T=$t;Start=$start;Size=$sz;Complete=($start+$sz -le $fs.Length)}
      $fs.Position = $start + $sz } } finally { $fs.Close() }
  $out }
function Report($f) {
  $a = Atoms $f
  $moov = @($a | Where-Object T -eq 'moov'); $moof = @($a | Where-Object T -eq 'moof').Count
  $last = $a[-1]
  "  [check A] moov atom present: $(if ($moov.Count -and $moov[0].Complete) {'YES (complete)'} else {'NO'})   moof fragments=$moof   last top-level atom=$($last.T) complete=$($last.Complete)"
  & $ff -hide_banner -v error -i $f -f null - 2>$null; "  [check B] full decode exit=$LASTEXITCODE"
  $dd = & $ffp -v error -show_entries 'format=duration' -of csv=p=0 $f 2>$null | Select-Object -First 1; $x=0.0
  if ([double]::TryParse("$dd".Trim(), [ref]$x)) {
    $tmp = Join-Path $S 'frame.png'; if (Test-Path $tmp) { Remove-Item $tmp }
    & $ff -hide_banner -v error -y -ss ($x-2) -i $f -frames:v 1 $tmp 2>$null; $e=$LASTEXITCODE
    $pb = if (Test-Path $tmp) { (Get-Item $tmp).Length } else { 0 }
    "  [check C] frame read 2 s before the end of the DATA (t=$([math]::Round($x-2,3))): exit=$e png_bytes=$pb"
  } else { "  [check C] frame read 2 s before end of DATA: no duration ($dd)" }
  "  [check D] row-6 end-check against the EXPECTED length (613.057 s):"
  & (Join-Path $S 'endcheck.ps1') -File $f -Expected 613.057 | ForEach-Object { "    $_" }
}
"# #142 row 6 truncation control, independent re-run $(Get-Date -Format s). Copies only; originals untouched."
foreach ($n in 'recording.original.mp4','recording.mp4') {
  $o = Join-Path $src $n; $h0 = (Get-FileHash $o -Algorithm SHA256).Hash
  "## ORIGINAL $n size=$((Get-Item $o).Length) sha256=$h0"
  Report $o
  foreach ($pct in 99, 90) {
    $c = Join-Path $S "copy-$pct-$n"; Copy-Item $o $c -Force
    $len = [int64][math]::Floor((Get-Item $c).Length * $pct / 100)
    $fs = [IO.File]::Open($c,'Open','ReadWrite'); $fs.SetLength($len); $fs.Close()
    "## COPY of $n truncated to $pct% -> size=$len"
    Report $c
  }
  $h1 = (Get-FileHash $o -Algorithm SHA256).Hash
  "## ORIGINAL $n sha256 after the run=$h1  unchanged=$($h0 -eq $h1)"
}
