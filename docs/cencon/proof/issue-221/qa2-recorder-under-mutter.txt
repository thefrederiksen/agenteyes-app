---- run.sh
#!/bin/bash
set -u; export DEBIAN_FRONTEND=noninteractive
grep PRETTY_NAME /etc/os-release
apt-get update -q >/dev/null 2>&1
apt-get install -y -q --no-install-recommends mutter xwayland dbus ffmpeg x11-xserver-utils libicu* ca-certificates pulseaudio >/tmp/i.log 2>&1; echo "apt exit=$?"
dpkg-query -W -f='${Package} ${Version}\n' mutter xwayland
export XDG_RUNTIME_DIR=/tmp/xdg HOME=/root; mkdir -p -m 700 $XDG_RUNTIME_DIR
unset DISPLAY WAYLAND_DISPLAY
timeout 300 dbus-run-session -- mutter --wayland --headless --virtual-monitor 1280x720 -- bash /inside.sh > /tmp/m.txt 2>&1; echo "mutter exit=$?"
grep -E 'Running Mutter|X11 display|^==|^exit=|^WAYLAND|^DISPLAY|ok\]|error|FAIL|refus|Wayland' /tmp/m.txt | grep -vE 'Gtk-WARNING|libEGL|MESA' | head -40
echo "== luminance of every png and mp4 written"
find / -xdev \( -name '*.png' -o -name '*.mp4' \) -path '/tmp/recordings/*' 2>/dev/null | while read f; do
  echo "$f ($(stat -c %s "$f") bytes)"
  ffmpeg -hide_banner -i "$f" -vf signalstats,metadata=print -f null - 2>&1 | grep -oE 'lavfi.signalstats.(YAVG|YMAX)=[0-9.]+' | sort | uniq -c | sort -rn | head -4
done
---- inside.sh (the client mutter starts)
#!/bin/bash
echo "== client env"; for v in WAYLAND_DISPLAY DISPLAY; do echo "$v=${!v-<unset>}"; done
cd /tmp
echo "== agenteyes screens"; timeout 60 /ae/agenteyes screens 2>&1 | grep -v glamor | head -6; echo "exit=${PIPESTATUS[0]}"
echo "== agenteyes shot --screen 1"; timeout 60 /ae/agenteyes shot --screen 1 > /tmp/shot.txt 2>&1; echo "exit=$?"; cat /tmp/shot.txt | tail -3
echo "== agenteyes video --screen 1 --seconds 3"; timeout 90 /ae/agenteyes video --screen 1 --seconds 3 > /tmp/vid.txt 2>&1; echo "exit=$?"; tail -4 /tmp/vid.txt
