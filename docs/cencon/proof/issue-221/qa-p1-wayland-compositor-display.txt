---- client script
#!/bin/bash
# Runs as the client command of a headless Wayland mutter. Prints what a client of the session sees.
echo "== client env as given by the compositor"
for v in WAYLAND_DISPLAY DISPLAY XDG_SESSION_TYPE XAUTHORITY; do echo "$v=${!v-<unset>}"; done
echo "== X sockets"; ls -la /tmp/.X11-unix/ 2>&1
echo "== xrandr against \$DISPLAY"; timeout 20 xrandr 2>&1 | head -4
for b in main branch; do
  echo "== agenteyes ($b) screens"; timeout 30 /ae-$b/agenteyes screens 2>&1 | head -8; echo "exit=$?"
done
echo "== ffmpeg x11grab of \$DISPLAY root (what the recorder records), no X11 client drawing"
timeout 30 ffmpeg -hide_banner -loglevel error -f x11grab -i "$DISPLAY" -frames:v 1 -y /out/root-empty.png; echo "ffmpeg exit=$?"
ffmpeg -hide_banner -i /out/root-empty.png -vf signalstats,metadata=print:key=lavfi.signalstats.YMAX -f null - 2>&1 | grep -o 'YMAX=[0-9]*' | head -1
echo "== a Wayland-native GTK window (GDK_BACKEND=wayland) and an X11 GTK window (GDK_BACKEND=x11)"
GDK_BACKEND=wayland gtk3-demo --run=pickers >/tmp/w.log 2>&1 & W=$!
sleep 6; timeout 30 ffmpeg -hide_banner -loglevel error -f x11grab -i "$DISPLAY" -frames:v 1 -y /out/root-with-wayland-client.png
ffmpeg -hide_banner -i /out/root-with-wayland-client.png -vf signalstats,metadata=print:key=lavfi.signalstats.YMAX -f null - 2>&1 | grep -o 'YMAX=[0-9]*' | head -1
echo "wayland client alive: $(kill -0 $W 2>/dev/null && echo yes || echo no)"; head -3 /tmp/w.log
GDK_BACKEND=x11 gtk3-demo --run=pickers >/tmp/x.log 2>&1 & X=$!
sleep 6; timeout 30 ffmpeg -hide_banner -loglevel error -f x11grab -i "$DISPLAY" -frames:v 1 -y /out/root-with-x11-client.png
ffmpeg -hide_banner -i /out/root-with-x11-client.png -vf signalstats,metadata=print:key=lavfi.signalstats.YMAX -f null - 2>&1 | grep -o 'YMAX=[0-9]*' | head -1
echo "x11 client alive: $(kill -0 $X 2>/dev/null && echo yes || echo no)"; head -3 /tmp/x.log
kill $W $X 2>/dev/null; echo "== inside done"

#!/bin/bash
echo "DISPLAY=${DISPLAY-<unset>} WAYLAND_DISPLAY=${WAYLAND_DISPLAY-<unset>}"
stat_y() { ffmpeg -hide_banner -i "$1" -vf signalstats,metadata=print -f null - 2>&1 | grep -oE 'Y(AVG|MAX)=[0-9.]+' | head -2 | tr '\n' ' '; echo " md5=$(md5sum < "$1" | cut -c1-12)"; }
for b in main branch; do
  echo "== agenteyes ($b) shot --screen 1, nothing open"; timeout 60 /ae-$b/agenteyes shot --screen 1 --out /out/shot-$b --label empty 2>&1 | tail -3; echo "exit=$?"
done
f=$(find /out/shot-branch -name '*.png' | head -1); echo "branch shot file: $f"; [ -n "$f" ] && stat_y "$f" && cp "$f" /out/A-empty.png
echo "== open a Wayland-native window (gtk3-demo, GDK_BACKEND=wayland)"
GDK_BACKEND=wayland gtk3-demo >/tmp/w.log 2>&1 & W=$!; sleep 8; echo "alive: $(kill -0 $W 2>/dev/null && echo yes || echo no)"
timeout 60 /ae-branch/agenteyes shot --screen 1 --out /out/shot-wl --label wl 2>&1 | tail -1
f=$(find /out/shot-wl -name '*.png' | head -1); [ -n "$f" ] && stat_y "$f" && cp "$f" /out/B-wayland-window.png
echo "== CONTROL: open an X11 window too (xclock -geometry 400x400+100+100 -bg white)"
xclock -geometry 400x400+100+100 -bg white >/tmp/x.log 2>&1 & X=$!; sleep 5; echo "alive: $(kill -0 $X 2>/dev/null && echo yes || echo no)"; head -2 /tmp/x.log

f=$(find /out/shot-x -name '*.png' | head -1); [ -n "$f" ] && stat_y "$f" && cp "$f" /out/C-wayland-plus-x11-window.png
echo "== agenteyes (branch) video --screen 1 --seconds 3"
timeout 90 /ae-branch/agenteyes video --screen 1 --seconds 3 --out /out/vid --label v 2>&1 | tail -4; echo "exit=$?"
find /out/vid -name '*.mp4' -exec ls -la {} \;
kill $W $X 2>/dev/null; echo "== inside2 done"
